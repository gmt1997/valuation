SQL para criar a tabela

CREATE TABLE IF NOT EXISTS valuation_models (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL DEFAULT 'Modelo Padrão',
  scenarios JSONB NOT NULL,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);
ALTER TABLE valuation_models ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all for anon" ON valuation_models FOR ALL USING (true) WITH CHECK (true);
